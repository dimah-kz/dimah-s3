---
title: Delete
description: Authorize deletion and run cleanup after S3 removes the object.
---

Enable with `delete: true` or an options object.

<Flow
  label="Delete flow"
  steps={[
    { name: "guard", kind: "hook" },
    { name: "DeleteObject", kind: "s3" },
    { name: "onDeleted", kind: "hook", note: "DB cleanup" },
  ]}
/>

```ts title="lib/s3.ts"
import { S3Client } from "@aws-sdk/client-s3";
import { dimahS3, DimahS3Error } from "@dimah-s3/server";

export const s3 = dimahS3({
  client: new S3Client({ /* env */ }),
  bucket: process.env.S3_BUCKET!,
  delete: {
    prefix: "uploads",
    guard: async ({ key, request }) => {
      if (!request.headers.get("authorization")) {
        throw new DimahS3Error("UNAUTHORIZED", { message: "Unauthorized" });
      }
      // then check that `key` may be deleted
    },
    onDeleted: async ({ key }) => {
      // object is already gone from S3 — drop the row for `key`
    },
  },
});
```

`onDeleted` is not a substitute for deleting from the bucket. It runs after.
