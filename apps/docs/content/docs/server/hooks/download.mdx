---
title: Download
description: Authorize before issuing a presigned GET URL.
---

Enable with `download: { enabled: true }`. The server never streams the file
— it signs a GET URL, then the browser fetches from S3.

```ts title="lib/s3.ts"
import { dimahS3, DimahS3Error } from "@dimah-s3/server";
import { s3Client, defaultBucket } from "@/lib/s3-client";

export const s3 = dimahS3({
  s3: s3Client,
  defaultBucket,
  download: {
    enabled: true,
    presignGuard: async ({ key, request }) => {
      if (!request.headers.get("authorization")) {
        throw new DimahS3Error("UNAUTHORIZED", { message: "Unauthorized" });
      }
      // then check that `key` is allowed
    },
  },
});
```

`onPresigned` receives the same fields plus `url` and `expiresIn`.
