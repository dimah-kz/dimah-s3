---
title: Download
description: Authorize before issuing a presigned GET URL.
---

Enable with `download: true` or an options object. The server never streams the
file — it signs a GET URL, then the browser fetches from S3.

  steps={[

```ts title="lib/s3.ts"
import { S3Client } from "@aws-sdk/client-s3";
import { dimahS3, DimahS3Error } from "@dimah-s3/server";

export const awsS3 = new S3Client({/* env */});

export const s3 = dimahS3({
  client: awsS3,
  bucket: process.env.S3_BUCKET!,
  download: {
    prefix: "uploads",
    guard: async ({ key, request }) => {
      if (!request.headers.get("authorization")) {
        throw new DimahS3Error("UNAUTHORIZED", { message: "Unauthorized" });
      }
      // then check that `key` is allowed
    },
  },
});
```

`onPresigned` receives the same fields plus `url` and `expiresIn`.
