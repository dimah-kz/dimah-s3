## Install

```npm
npm i @dimah-s3/react
```

### API client [step]

Hooks talk to an `S3Api`. `createS3Client` returns the client, a bound
provider, and a typed `useApi`.

<Tabs items={["@dimah-s3/server", "Custom backend"]}>

<Tab value="@dimah-s3/server">

If the server is mounted at `/api/s3` ([Server setup](/docs/server/setup)):

```tsx title="components/s3-provider.tsx"
"use client";

import { createS3Client } from "@dimah-s3/react";

export const { api, S3Provider, useApi } = createS3Client();

export function S3ClientProvider({ children }: { children: React.ReactNode }) {
  return <S3Provider>{children}</S3Provider>;
}
```

Custom base path, plugins, or auth:

```ts
import { createS3Client } from "@dimah-s3/react";
import { dbClient } from "@dimah-s3/db/client";

export const { api, S3Provider, useApi } = createS3Client({
  basePath: "/api/your-path",
  plugins: [dbClient()],
  credentials: "include",
  headers: async () => ({
    Authorization: `Bearer ${await getAccessToken()}`,
  }),
});
```

`basePath` must match the server. Failed API calls throw `DimahS3Error`
(`.code`, `.status`). Call `api` directly, or `useApi()` inside React.

</Tab>

<Tab value="Custom backend">

Implement `S3Api` with `defineApi` and pass it to the provider. Payload
shapes: [Custom backend](/docs/react/custom-backend).

```tsx title="components/s3-provider.tsx"
"use client";

import { S3Provider } from "@dimah-s3/react";
import { api } from "@/lib/s3-api";

export function S3ClientProvider({ children }: { children: React.ReactNode }) {
  return <S3Provider api={api}>{children}</S3Provider>;
}
```

### Provider [step]

Mount once near the app root. Hooks inside can omit `api`.

<Tabs items={["Next.js (App Router)", "Other React apps"]}>

<Tab value="Next.js (App Router)">

```tsx title="app/layout.tsx"
import { S3ClientProvider } from "@/components/s3-provider";

export default function RootLayout({
  children,
}: {
  children: React.ReactNode;
}) {
  return (
    <html lang="en">
      <body>
        <S3ClientProvider>{children}</S3ClientProvider>
      </body>
    </html>
  );
}
```

</Tab>

<Tab value="Other React apps">

```tsx title="src/main.tsx"
import React from "react";
import ReactDOM from "react-dom/client";
import { S3ClientProvider } from "./components/s3-provider";
import { App } from "./app";

ReactDOM.createRoot(document.getElementById("root")!).render(
  <React.StrictMode>
    <S3ClientProvider>
      <App />
    </S3ClientProvider>
  </React.StrictMode>,
);
```

### First hook [step]

```tsx title="app/uploader.tsx"
"use client";

import { useUpload } from "@dimah-s3/react";

export function Uploader() {
  const { getRootProps, getInputProps, isUploading, progress } = useUpload({
    objectKey: "uploads/photo.jpg",
  });

  return (
    <div {...getRootProps()}>
      <input {...getInputProps()} />
      {isUploading ? `${progress.percent}%` : "Drop a file or click to upload"}
    </div>
  );
}
```

Need a finished button instead? [Upload Button](/docs/react/ui/components/upload-button).
