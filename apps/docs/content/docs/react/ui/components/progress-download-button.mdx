import { ProgressDownloadDemo } from "@/components/demos/progress-download-demo";

`ProgressDownloadButton` downloads through fetch so it can show progress and
support cancel. Prefer [`DownloadButton`](/docs/react/ui/components/download-button) when
you only need a simple native download.

## Install

One-time styles and toaster: [Setup](/docs/react/ui/setup).

<Tabs items={["npm package", "shadcn registry"]}>

<Tab value="npm package">

```tsx

```

</Tab>

<Tab value="shadcn registry">

```npm
npx shadcn@latest add @dimah-s3/progress-download-button
```

</Tab>

</Tabs>

## Usage

```tsx
<ProgressDownloadButton objectKey="uploads/report.pdf" size="sm" />
```

Errors render as a [`StatusAttachment`](/docs/react/ui/components/attachment) row
(`attachmentSize` / `attachmentOrientation`).
