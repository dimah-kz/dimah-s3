import { ObjectListDemo } from "@/components/demos/object-list-demo";

`ObjectList` renders a folder catalog. Pass the object returned by
[`useObjectList`](/docs/react/hooks/object-list) as `list`. The component
does not fetch and does not call the hook. A server payload or your own
store works when it has the same fields.

`query` for your API, or for `dbObjectQuery`, without changing the component.

## Usage

```tsx
"use client";

import {
  useDelete,
  useDownload,
  useObjectList,
  type ObjectListQuery,
} from "@dimah-s3/react";
import { ObjectList, ObjectListMenu } from "@dimah-s3/ui";

type LibraryFile = {
  id: string;
  name: string;
  key: string;
  sizeLabel: string;
  type: string;
};

export function Library({ query }: { query: ObjectListQuery<LibraryFile> }) {
  const list = useObjectList({ query });
  const download = useDownload({ route: "docs" });
  const share = useDownload({ route: "docs", disposition: "inline" });
  const remove = useDelete({ route: "docs" });

  return (
    <ObjectList
      list={list}
      getKey={(file) => file.id}
      getLabel={(file) => file.name}
      getDescription={(file) => file.sizeLabel}
      getFileType={(file) => file.type}
      download={download}
      delete={remove}
      getObjectKey={(file) => file.key}
      renderActions={(file) => (
        <ObjectListMenu
          objectKey={file.key}
          fileName={file.name}
          download={download}
          share={share}
          delete={remove}
        />
      )}
    />
  );
}
```

`getKey` and `getLabel` are the only required accessors. Pass `download` or `delete` to select the loaded rows. The checkbox sits inside the row. Shift-click a checkbox to extend from the previous row. Escape clears the selection. The header then switches to a bar with download and delete. `download` is navigate-mode `useDownload`. `getObjectKey` is the stored object key when it differs from `getKey`. `resolveFolderKeys` supplies the keys inside a checked folder; without it, checked folders stay in the list and are not downloaded or deleted. `renderActions` is where the row menu goes. `share` is a second `useDownload` with `disposition: "inline"` so Copy link opens in a browser while Download stays an attachment.

`trail` drives the breadcrumb. `useObjectList` fills it when you open folders with `openFolder` and `navigate`. For a controlled URL, keep the same path in `defaultTrail` on the first render, or pass `trail` on the `list` object.
