## Usage

```tsx
import { FileAttachment, StatusAttachment } from "@dimah-s3/ui";

// Uploading avatar row
<FileAttachment
  state="uploading"
  fileName="avatar.png"
  fileType="image/png"
  fileSize={180_000}
  percent={65}
  previewUrl="/dimah-avatar.jpg"
/>

// Generic status row
<StatusAttachment state="processing" title="Presigning URL…" />
```
