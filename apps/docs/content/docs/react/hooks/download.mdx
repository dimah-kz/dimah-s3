---
title: useDownload
description: Presign GET URLs for direct download, inline preview, or progress tracking.
---

`useDownload` requests a signed GET URL from your API and manages browser navigation, fetch streams, and inline previews.

```tsx
"use client";

import { useDownload } from "@dimah-s3/react";

export function DownloadAvatar({ avatarKey }: { avatarKey: string }) {
  const { download, phase } = useDownload({ route: "avatar" });

  return (
    <button type="button" onClick={() => void download(avatarKey)}>
      {phase === "presigning" ? "Generating link…" : "Download Avatar"}
    </button>
  );
}
```

---

## Preview with `useObjectUrl`

For inline preview in `<img>`, `<video>`, or `<iframe>`:

```tsx
"use client";

import { useObjectUrl } from "@dimah-s3/react";

export function AvatarImage({ avatarKey }: { avatarKey: string }) {
  const { url, isLoading } = useObjectUrl({
    route: "avatar",
    objectKey: avatarKey,
    disposition: "inline",
  });

  if (isLoading) return <div>Loading...</div>;
  if (!url) return null;

  return (
    <img
      src={url}
      alt="User Avatar"
      className="size-16 rounded-full object-cover"
    />
  );
}
```

## Download modes

| Mode                 | Hook Usage                                        | Description                                             |
| -------------------- | ------------------------------------------------- | ------------------------------------------------------- |
| `navigate` (default) | `useDownload({ route: "avatar" })`                | Fast browser navigation directly to presigned URL       |
| `fetch`              | `useDownload({ route: "avatar", mode: "fetch" })` | Streams bytes with live progress and cancel support     |
| `inline`             | `useObjectUrl({ route: "avatar", objectKey })`    | Fetches signed inline URL with automatic expiry caching |

## Frequently asked questions

<Accordions>

<Accordion title="How do I get just the presigned URL without triggering a download?">

Use the `presign` method from `useDownload`:

```tsx
const { presign } = useDownload({ route: "avatar", disposition: "inline" });

const result = await presign(avatarKey);
console.log("Signed URL:", result.url);
```

</Accordion>

<Accordion title="How do I show download progress with a cancel button?">

Set `mode: "fetch"`:

```tsx
const { download, cancel, progress, isDownloading } = useDownload({
  route: "document",
  mode: "fetch",
});

return (
  <div>
    <button onClick={() => download(documentKey)}>
      {isDownloading ? `${progress.percent}%` : "Download"}
    </button>
    {isDownloading && <button onClick={cancel}>Cancel</button>}
  </div>
);
```
